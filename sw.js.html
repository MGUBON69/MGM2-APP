// กำหนดชื่อแคชใหม่เป็น v3 เพื่อบังคับให้มือถือทุกเครื่องลบหน้าเก่าแล้วโหลดหน้าใหม่ทันที const CACHE_NAME = 'mg-ubon-app-v3'; const ASSETS_TO_CACHE = [ 'index.html', 'index2.html', 'manifest.json', 'MG_profile.png' ]; // สถาปัตยกรรมการติดตั้งระบบควบคุมแคชส่วนหน้า self.addEventListener('install', (event) => { event.waitUntil( caches.open(CACHE_NAME).then((cache) => { return cache.addAll(ASSETS_TO_CACHE); }) ); }); // สั่งให้ Service Worker ตัวใหม่ทำงานทันทีโดยไม่ต้องรอปิดแอปเก่า self.addEventListener('activate', (event) => { event.waitUntil( caches.keys().then((cacheNames) => { return Promise.all( cacheNames.map((cache) => { if (cache !== CACHE_NAME) { console.log('ล้างแคชเวอร์ชันเก่าเรียบร้อยแล้ว'); return caches.delete(cache); } }) ); }) ); }); // สถาปัตยกรรมจัดการคำขอข้อมูล self.addEventListener('fetch', (event) => { event.respondWith( caches.match(event.request).then((cachedResponse) => { return cachedResponse || fetch(event.request); }) ); });
